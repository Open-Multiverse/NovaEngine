//! 集成测试 - 角色移动系统（MOVE-01 至 MOVE-04）

use bevy::prelude::*;
use nova_character::{
    attributes::Attributes,
    character::CharacterBundle,
    state::CharacterState,
    NovaCharacterPlugin,
};
use nova_map::tilemap::TileMap;
use nova_test::TestApp;

/// MOVE-01：Moving 状态下 Transform 每帧向 target 更新
#[test]
fn test_movement_updates_transform() {
    let mut app = TestApp::new();
    app.add_plugin(NovaCharacterPlugin);

    let target = Vec3::new(10.0, 0.0, 0.0);
    let start = Vec3::ZERO;

    let entity = app
        .world_mut()
        .spawn(CharacterBundle {
            state: CharacterState::Moving { target },
            transform: Transform::from_translation(start),
            ..default()
        })
        .id();

    // 运行若干帧（无 TileMap，直线降级模式）
    app.run_frames(10);

    let transform = app.world().get::<Transform>(entity).unwrap();
    let distance_to_target = (transform.translation - target).length();
    let initial_distance = (start - target).length();

    // 单位应已向 target 靠近（不在原点）
    assert!(
        transform.translation.x > 0.1,
        "单位应已向 target 移动，当前 x = {}",
        transform.translation.x
    );
    // 距离应比初始更近
    assert!(
        distance_to_target < initial_distance,
        "当前距离 {} 应小于初始距离 {}",
        distance_to_target,
        initial_distance
    );
}

/// MOVE-02：单位到达目标后自动切换为 Idle
#[test]
fn test_movement_state_transition_to_idle() {
    let mut app = TestApp::new();
    app.add_plugin(NovaCharacterPlugin);

    // 目标非常近，几帧内必然到达
    let target = Vec3::new(0.1, 0.0, 0.0);

    let entity = app
        .world_mut()
        .spawn(CharacterBundle {
            state: CharacterState::Moving { target },
            transform: Transform::from_translation(Vec3::ZERO),
            ..default()
        })
        .id();

    // 运行足够多帧确保到达（target 距离 0.1 < 0.5 阈值，首帧即应切换）
    app.run_frames(5);

    let state = app.world().get::<CharacterState>(entity).unwrap();
    assert!(
        matches!(state, CharacterState::Idle),
        "到达目标后状态应为 Idle，当前状态为 {:?}",
        state
    );
}

/// MOVE-03：移动速度由 Attributes.move_speed 驱动
#[test]
fn test_movement_speed_from_attributes() {
    // 场景 A：speed = 2.0（慢）
    let position_slow = {
        let mut app = TestApp::new();
        app.add_plugin(NovaCharacterPlugin);

        let entity = app
            .world_mut()
            .spawn(CharacterBundle {
                state: CharacterState::Moving {
                    target: Vec3::new(100.0, 0.0, 0.0),
                },
                transform: Transform::from_translation(Vec3::ZERO),
                attributes: Attributes {
                    move_speed: 2.0,
                    ..default()
                },
                ..default()
            })
            .id();

        app.run_frames(5);
        app.world().get::<Transform>(entity).unwrap().translation.x
    };

    // 场景 B：speed = 10.0（快）
    let position_fast = {
        let mut app = TestApp::new();
        app.add_plugin(NovaCharacterPlugin);

        let entity = app
            .world_mut()
            .spawn(CharacterBundle {
                state: CharacterState::Moving {
                    target: Vec3::new(100.0, 0.0, 0.0),
                },
                transform: Transform::from_translation(Vec3::ZERO),
                attributes: Attributes {
                    move_speed: 10.0,
                    ..default()
                },
                ..default()
            })
            .id();

        app.run_frames(5);
        app.world().get::<Transform>(entity).unwrap().translation.x
    };

    assert!(
        position_fast > position_slow,
        "speed=10 移动距离 ({}) 应大于 speed=2 移动距离 ({})",
        position_fast,
        position_slow
    );
}

/// MOVE-04：有 TileMap 时，单位沿 PathFollow 路径点移动
#[test]
fn test_pathfollow_integration() {
    let mut app = TestApp::new();
    app.add_plugin(NovaCharacterPlugin);

    // 创建一个 10x10 的全草地地图（全部可通行），瓦片尺寸 1.0
    // 地图范围：世界坐标 -5.0 到 5.0（X/Z 轴）
    let tilemap = TileMap::new(10, 10, 1.0);
    app.world_mut().insert_resource(tilemap);

    // 单位从地图左侧（世界坐标约 -4.0, 0, -4.0）出发
    // 目标在地图右侧（世界坐标约 4.0, 0, 4.0）
    // world_to_tile(-4.0, 0, -4.0) → 约 (0, 0)
    // world_to_tile(4.0, 0, 4.0) → 约 (9, 9)
    let start_world = Vec3::new(-4.0, 0.0, -4.0);
    let target_world = Vec3::new(4.0, 0.0, 4.0);

    let entity = app
        .world_mut()
        .spawn(CharacterBundle {
            state: CharacterState::Moving { target: target_world },
            transform: Transform::from_translation(start_world),
            ..default()
        })
        .id();

    // 运行 2 帧：第 1 帧触发寻路并插入 PathFollow（Commands 延迟生效），
    // 第 2 帧 PathFollow 可用，movement_system 开始路径跟随
    app.run_frames(2);

    // 验证 PathFollow 已被插入（说明 A* 寻路成功）
    let has_path_follow = app.world().get::<nova_map::pathfinding::PathFollow>(entity).is_some();
    assert!(
        has_path_follow,
        "有 TileMap 时，首次进入 Moving 应触发 A* 寻路并插入 PathFollow 组件"
    );

    // 继续运行足够多帧，单位应向目标移动
    app.run_frames(30);

    let transform = app.world().get::<Transform>(entity).unwrap();
    // 单位应已离开起点（向 target 方向移动）
    let moved_distance = (transform.translation - start_world).length();
    assert!(
        moved_distance > 0.5,
        "单位应已沿路径移动，移动距离 {} 应大于 0.5",
        moved_distance
    );
}
